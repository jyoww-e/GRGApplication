1. 概述

库信息

库名称YTZN_DevBase
WindowsYTZN_DevBase.dll / YTZN_DevBaseD.dll (Debug)
LinuxlibYTZN_DevBase.so/ libYTZN_DevBaseD.so(Debug)
导入库YTZN_DevBase.lib / YTZN_DevBaseD.lib

主要功能

  • 🔌 串口(COM)通讯 — 支持配置、发送、接收等完整操作
  • 🔗 USB通讯 — 支持 HID 设备的数据收发
  • 📄 配置文件管理 — ini格式配置文件读写/win注册表方式读写
  • 📝 日志功能 — 支持分级日志输出
  • 🌐 跨平台 — Windows / Linux 统一接口

2. 接口说明

2.1 设备初始化

void vSetLogicalDevName(const char* p_pcLogicalDevName);

设置逻辑设备名称，用于日志标识。

void vGetLogicalDevName(char* p_pcLogicalDevName);

获取逻辑设备名称。

iPreInitDev 跨平台

int iPreInitDev(bool p_bIsExePath = true);

初始化设备，加载配置文件 GRGDTATM_CommCfg.ini。

p_bIsExePathbooltrue: 从exe目录读取配置
false: 从DevBase库目录读取配置，开源鸿蒙场景是在/data/app/sxd/dunite目录下读取
返回值: SUCCESS(0) 成功 | E_DEVBASE_LOGICNAME_NULL(920) | E_DEVBASE_OPEN_COMCFGFILE(921)

2.2 通讯配置获取

iGetDefaultCommConfig 仅支持串口

int iGetDefaultCommConfig(tCommCfg& p_sCommCfg);

获取默认串口(COM)通讯配置。

iGetDefaultCommConfig2 支持USB和串口配置

int iGetDefaultCommConfig2(tCommCfg2& p_sCommCfg);

获取默认通讯配置(支持USB扩展)。

2.3 通讯初始化

iInitComm 仅支持串口

int iInitComm(tCommCfg p_sCommCfg, bool p_bEnableEvents = false);

初始化串口(COM)通讯。

p_sCommCfgtCommCfg串口配置参数
p_bEnableEventsbool是否启用事件模式(默认false)

iInitComm2 支持USB和串口配置

int iInitComm2(tCommCfg2 p_sCommCfg, bool p_bEnableEvents = false);

初始化USB通讯。

p_sCommCfgtCommCfg2USB配置参数
p_bEnableEventsbool是否启用事件模式(默认false)

2.4 串口数据收发

iSendCommand 仅串口

int iSendCommand(const BYTE* p_pbySendCmd, UINT p_uiCmdLength, bool p_bClearRecvBuf = true, bool p_bLog = true);

通过串口发送命令。

p_pbySendCmdconst BYTE*发送数据缓冲区
p_uiCmdLengthUINT发送数据长度
p_bClearRecvBufbool发送前是否清空接收缓冲区
p_bLogbool是否记录日志
返回值: SUCCESS(0) | E_DEVBASE_INVALID_PARA(952) | E_DEVBASE_ACCESS_COM(972) | E_DEVBASE_SEND_CMD(973)

iWaitResult 仅串口

int iWaitResult(BYTE* p_pbyDataRecv, UINT& p_uiRecvLength, UINT p_uiTimeout, bool p_bLog = true);

等待并接收串口数据（非阻塞模式）。

p_pbyDataRecvBYTE*接收数据缓冲区
p_uiRecvLengthUINT&输入: 缓冲区大小 | 输出: 实际接收长度
p_uiTimeoutUINT超时时间(毫秒), 最大60000ms
p_bLogbool是否记录日志
返回值: SUCCESS(0) | E_DEVBASE_INVALID_PARA(952) | E_DEVBASE_RECV_DATA(974) | E_DEVBASE_ACCESS_COM(972)

iSendAndRecvData 仅串口

int iSendAndRecvData(const BYTE* p_pbySendCmd, UINT p_uiCmdLength, BYTE* p_pbyDataRecv, UINT& p_uiRecvLength, UINT p_uiTimeout, bool p_bClearRecvBuf = true, bool p_bLog = true);

发送命令并等待响应（串口，一站式调用）。

USB与串口数据接收

p_pbyDataRecvBYTE*接收数据缓冲区
p_uiRecvLengthUINT&输入: 缓冲区大小 | 输出: 实际接收长度

p_bLogbool是否记录日志

iSendAndRecvData2 支持USB和串口配置

3. 结构体说明

通讯类型枚举

typedef enum __MIDL_IComm_0001
{
    COMM_COM = 1,       // 串口
    COMM_COMBOARD = 2,  // 串口扩展
    COMM_USB = 3,       // USB
    COMM_TCPIP = 4      // TCP/IP
} tCommType;

串口信息结构体

struct tComInfo
{
    unsigned int iDatabits;       // 数据位 (5,6,7,8)
    unsigned int iStopbits;       // 停止位 (ONESTOPBIT=0, TWOSTOPBITS=2)
    unsigned int iParity;         // 校验位 (NOPARITY=0, ODDPARITY=1, EVENPARITY=2)
    unsigned int iComID;          // COM端口号
    unsigned int iBaudRate;       // 波特率
    unsigned int iFlowcontrol;    // 流控制 (CTSFLOW=1, RTSFLOW=2)
    
    // 超时参数
    unsigned int ReadIntervalTimeout;         // 读取间隔超时(毫秒)
    unsigned int ReadTotalTimeoutMultiplier;   // 读取超时乘数
    unsigned int ReadTotalTimeoutConstant;    // 读取固定超时(毫秒)
    // ... 其他扩展参数
};

USB信息结构体

struct tUSBInfo2
{
    GUID sGuid;                  // 设备GUID
    char acSendPipes[50];       // 发送管道字符串
    char acRecvPipes[50];       // 接收管道字符串
    char acExtendStr[256];      // 扩展字符串
    int iReserve;                // 保留字段
};

错误返回结构体

typedef struct
{
    int  iLogicCode;          // 逻辑错误码
    int  iPhyCode;            // 物理错误码
    int  iHandle;             // 句柄
    int  iType;              // 类型
    char acDevReturn[128];    // 错误描述字符串
    char acReserve[128];      // 保留字段
} tDevReturn;

6. 调用示例

6.1 串口通讯完整示例

COM串口通讯
#include "DevBase.h"
#include 

int main()
{
    // 1. 创建设备对象
    YTZN_ODevBase* pDev = new YTZN_ODevBase();
    
    // 2. 设置逻辑设备名称
    pDev->vSetLogicalDevName("TestDev");
    
    // 3. 初始化设备(从exe目录读取配置)
    int iRet = pDev->iPreInitDev(true);
    if (SUCCESS != iRet)
    {
        printf("iPreInitDev failed: %d\n", iRet);
        delete pDev;
        return -1;
    }
    
    // 4. 获取默认串口配置
    tCommCfg sCommCfg;
    iRet = pDev->iGetDefaultCommConfig(sCommCfg);
    if (SUCCESS != iRet)
    {
        printf("iGetDefaultCommConfig failed: %d\n", iRet);
        delete pDev;
        return -1;
    }
    
    // 5. 修改配置参数
    sCommCfg.unCommInfo.sComInfo.iBaudRate = 9600;
    sCommCfg.unCommInfo.sComInfo.iComID = 3;  // COM3
    
    // 6. 初始化串口通讯
    iRet = pDev->iInitComm(sCommCfg);
    if (SUCCESS != iRet)
    {
        printf("iInitComm failed: %d\n", iRet);
        delete pDev;
        return -1;
    }
    
    // 7. 发送命令并等待响应
    BYTE bySendData[] = {0x01, 0x03, 0x00, 0x00, 0x00, 0x0A};
    BYTE byRecvData[256];
    UINT uiRecvLen = sizeof(byRecvData);
    
    iRet = pDev->iSendAndRecvData(
        bySendData, sizeof(bySendData),
        byRecvData, uiRecvLen,
        1000,    // 超时1000ms
        true,    // 发送前清空接收缓冲
        true     // 记录日志
    );
    
    if (SUCCESS == iRet)
    {
        printf("Receive %d bytes\n", uiRecvLen);
    }
    
    // 8. 关闭串口
    pDev->iCloseComm();
    
    // 9. 释放资源
    delete pDev;
    
    return 0;
}

6.2 USB通讯完整示例

USB通讯
#include "DevBase.h"
#include 

int main()
{
    // 1. 创建设备对象
    YTZN_ODevBase* pDev = new YTZN_ODevBase();
    
    // 2. 设置逻辑设备名称
    pDev->vSetLogicalDevName("USB_Dev");
    
    // 3. 初始化设备(从库目录读取配置)
    int iRet = pDev->iPreInitDev(false);
    if (SUCCESS != iRet)
    {
        printf("iPreInitDev failed: %d\n", iRet);
        delete pDev;
        return -1;
    }
    
    // 4. 获取默认USB配置
    tCommCfg2 sCommCfg;
    iRet = pDev->iGetDefaultCommConfig2(sCommCfg);
    if (SUCCESS != iRet)
    {
        printf("iGetDefaultCommConfig2 failed: %d\n", iRet);
        delete pDev;
        return -1;
    }
    
    // 5. 设置USB配置
    sCommCfg.eCommType = COMM_USB;
    // ... 设置GUID和管道信息
    
    // 6. 初始化USB通讯
    iRet = pDev->iInitComm2(sCommCfg);
    if (SUCCESS != iRet)
    {
        printf("iInitComm2 failed: %d\n", iRet);
        delete pDev;
        return -1;
    }
    
    // 7. 发送命令
    BYTE bySendData[] = {0x01, 0x02, 0x03, 0x04};
    iRet = pDev->iSendCommand2(bySendData, sizeof(bySendData), 0x01, true, true);
    
    // 8. 等待响应
    BYTE byRecvData[256];
    UINT uiRecvLen = sizeof(byRecvData);
    iRet = pDev->iWaitResult2(byRecvData, uiRecvLen, 1000, 0x81, true);
    
    // 9. 关闭USB
    pDev->iCloseComm2();
    
    // 10. 释放资源
    delete pDev;
    
    return 0;
}

6.3 配置读写示例

INI配置读写
#include "DevBase.h"

// 读取配置
void readConfig(YTZN_ODevBase* pDev)
{
    int iValue = 0;
    char acStrValue[256] = {0};
    UINT uiLen = sizeof(acStrValue);
    
    // 读取整数值
    if (SUCCESS == pDev->iGetIntValue("COMSetting", "BaudRate", iValue))
    {
        printf("BaudRate: %d\n", iValue);
    }
    
    // 读取字符串值
    if (SUCCESS == pDev->iGetStringValue("LogicalName", "DevName", acStrValue, uiLen))
    {
        printf("DevName: %s\n", acStrValue);
    }
}

// 写入配置
void writeConfig(YTZN_ODevBase* pDev)
{
    pDev->iSetIntValue("COMSetting", "BaudRate", 9600);
    
    char acStr[] = "NewDevName";
    pDev->iSetStringValue("LogicalName", "DevName", acStr, strlen(acStr));
}

6.4 错误处理示例

错误处理
#include "DevBase.h"

void handleError(YTZN_ODevBase* pDev)
{
    // 获取错误标志
    int iErrorFlag = pDev->iGetErrorFlag();
    if (iErrorFlag != 0)
    {
        printf("Hardware Error Flag: 0x%08X\n", iErrorFlag);
    }
    
    // 获取详细错误信息
    tDevReturn sDevErr = pDev->tGetErrorFlag();
    printf("Logic Code: %d\n", sDevErr.iLogicCode);
    printf("Physical Code: %d\n", sDevErr.iPhyCode);
    printf("Error Message: %s\n", sDevErr.acDevReturn);
}
